cmake_minimum_required(VERSION 3.21)
project(WeldSeamScheme1 VERSION 4.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(WELD_SEAM_BUILD_APPS "Build local GUI/CLI tools (not installed with SDK)" OFF)
option(BUILD_TESTING "Build algorithm and SDK regression tests" OFF)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/install" CACHE PATH "SDK install directory" FORCE)
endif()
include(CMakePackageConfigHelpers)
set(PCL_DIR "E:/environments/PCL1.11.0/cmake" CACHE PATH "PCL package directory")
find_package(PCL 1.11 REQUIRED COMPONENTS common filters)
find_path(PCL_FLANN_INCLUDE_DIR flann/util/params.h
  HINTS ${FLANN_INCLUDE_DIRS} ${FLANN_INCLUDE_DIR} "${FLANN_ROOT}/include"
  DOC "FLANN headers (PCL 1.11 Windows targets do not propagate them)")
if(NOT PCL_FLANN_INCLUDE_DIR)
  message(FATAL_ERROR "FLANN headers required by the PCL algorithm modules were not found")
endif()

# Memory-in / memory-out library. No PCL io, Qt or VTK in this target.
add_library(seam_core SHARED src/SeamDetector.cpp src/UniformSeamWidth.cpp
  src/ProfileCurvature.cpp src/HullSurfaces.cpp src/SeamSmoother.cpp
  src/SeamHistory.cpp src/ConfigIO.cpp src/WeldSeam.cpp src/SurfaceClouds.cpp src/BSplineSurface.cpp)
set_target_properties(seam_core PROPERTIES OUTPUT_NAME WeldSeam EXPORT_NAME WeldSeam
  WINDOWS_EXPORT_ALL_SYMBOLS ON DEBUG_POSTFIX d AUTOMOC OFF)
add_library(WeldSeam::WeldSeam ALIAS seam_core)
target_include_directories(seam_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
  $<INSTALL_INTERFACE:include/WeldSeam>)
target_include_directories(seam_core SYSTEM PUBLIC "$<BUILD_INTERFACE:${PCL_INCLUDE_DIRS}>")
target_include_directories(seam_core PRIVATE third_party)
target_include_directories(seam_core SYSTEM PRIVATE ${PCL_FLANN_INCLUDE_DIR})
target_compile_features(seam_core PUBLIC cxx_std_14)
target_link_libraries(seam_core PUBLIC pcl_common pcl_filters)
set(_weld_targets seam_core)

if(WELD_SEAM_BUILD_APPS)
  set(Qt5_DIR "E:/environments/qt5/5.14.1/msvc2017_64/lib/cmake/Qt5" CACHE PATH "Qt package directory")
  find_package(Qt5 5.14 REQUIRED COMPONENTS Core Gui Widgets Concurrent)
  find_package(PCL 1.11 REQUIRED COMPONENTS common filters io visualization)
  set(CMAKE_AUTOMOC ON)
  add_library(seam_result_io STATIC src/ResultIO.cpp src/PointCloudIO.cpp)
  target_link_libraries(seam_result_io PUBLIC seam_core Qt5::Core pcl_io)
  add_library(seam_debug_viewer STATIC src/SeamDebugViewer.cpp)
  target_link_libraries(seam_debug_viewer PUBLIC seam_core Qt5::Widgets pcl_visualization ${VTK_LIBRARIES} vtkGUISupportQt)
  add_executable(weld_seam_app WIN32 src/main.cpp src/MainWindow.cpp src/MainWindow.h)
  target_link_libraries(weld_seam_app PRIVATE seam_debug_viewer seam_result_io Qt5::Widgets Qt5::Concurrent pcl_visualization ${VTK_LIBRARIES} vtkGUISupportQt)
  add_executable(weld_seam_cli src/cli_main.cpp)
  target_link_libraries(weld_seam_cli PRIVATE seam_debug_viewer seam_result_io Qt5::Core)
  set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT weld_seam_app)
  list(APPEND _weld_targets seam_result_io seam_debug_viewer weld_seam_app weld_seam_cli)
  if(BUILD_TESTING)
    add_executable(result_io_test tests/test_result_io.cpp)
    target_link_libraries(result_io_test PRIVATE seam_result_io)
    add_test(NAME result_pcd_roundtrip COMMAND result_io_test)
    list(APPEND _weld_targets result_io_test)
  endif()
endif()

if(BUILD_TESTING)
  enable_testing()
  # Existing regressions read measured TXT fixtures. This helper is not part of
  # the SDK and deliberately does not compile PCD/PLY backends or link pcl_io.
  add_library(seam_test_io STATIC src/PointCloudIO.cpp)
  target_compile_definitions(seam_test_io PRIVATE WELD_SEAM_TEXT_IO_ONLY)
  target_link_libraries(seam_test_io PUBLIC seam_core)
  add_executable(seam_detector_test tests/test_detector.cpp)
  target_link_libraries(seam_detector_test PRIVATE seam_core seam_test_io)
  target_compile_definitions(seam_detector_test PRIVATE SEAM_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  add_test(NAME synthetic_seam_detection COMMAND seam_detector_test)
  add_executable(config_io_test tests/test_config_io.cpp)
  target_link_libraries(config_io_test PRIVATE seam_core)
  target_compile_definitions(config_io_test PRIVATE SEAM_DEFAULT_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/weld_seam.json")

  add_executable(cloud_api_test tests/test_cloud_api.cpp)
  target_link_libraries(cloud_api_test PRIVATE seam_core seam_test_io)
  target_compile_definitions(cloud_api_test PRIVATE SEAM_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  add_test(NAME cloud_only_api COMMAND cloud_api_test)
  add_executable(surface_clouds_test tests/test_surface_clouds.cpp)
  target_link_libraries(surface_clouds_test PRIVATE seam_core seam_test_io)
  target_compile_definitions(surface_clouds_test PRIVATE SEAM_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  add_test(NAME three_surface_clouds COMMAND surface_clouds_test)
  add_executable(bspline_surface_test tests/test_bspline_surface.cpp)
  target_link_libraries(bspline_surface_test PRIVATE seam_core)
  add_test(NAME bspline_surface_fitting COMMAND bspline_surface_test)
  list(APPEND _weld_targets bspline_surface_test)
  add_executable(profile_spline_test tests/test_profile_spline.cpp)
  target_link_libraries(profile_spline_test PRIVATE seam_core)
  add_test(NAME robust_profile_spline COMMAND profile_spline_test)
  list(APPEND _weld_targets profile_spline_test)
  add_executable(seam_diagnostic tests/detect_diagnostic.cpp)
  target_link_libraries(seam_diagnostic PRIVATE seam_core seam_test_io)
  list(APPEND _weld_targets seam_test_io seam_detector_test config_io_test cloud_api_test surface_clouds_test seam_diagnostic)
endif()

if(MSVC)
  foreach(target IN LISTS _weld_targets)
    target_compile_options(${target} PRIVATE /utf-8 /bigobj /MP4)
    target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:/Zi>)
  endforeach()
  if(WELD_SEAM_BUILD_APPS)
    foreach(target weld_seam_app weld_seam_cli)
      target_link_options(${target} PRIVATE $<$<CONFIG:Release>:/DEBUG>)
    endforeach()
  endif()
endif()

# Only the library is deliverable; executables remain local development tools.
install(TARGETS seam_core EXPORT WeldSeamTargets
  RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES src/WeldSeam.h src/ConfigIO.h src/SeamTypes.h src/AlgorithmParameters.h src/BSplineSurface.h
  DESTINATION include/WeldSeam)
install(EXPORT WeldSeamTargets NAMESPACE WeldSeam:: DESTINATION lib/cmake/WeldSeam)
configure_package_config_file(cmake/WeldSeamConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/WeldSeamConfig.cmake" INSTALL_DESTINATION lib/cmake/WeldSeam)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/WeldSeamConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/WeldSeamConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/WeldSeamConfigVersion.cmake" DESTINATION lib/cmake/WeldSeam)
install(CODE "if(NOT EXISTS \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/bin/config/weld_seam.json\")\n file(INSTALL DESTINATION \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/bin/config\" TYPE FILE FILES \"${CMAKE_CURRENT_SOURCE_DIR}/config/weld_seam.json\")\nendif()")
install(FILES config/weld_seam.json DESTINATION bin/config RENAME weld_seam.defaults.json)
install(DIRECTORY examples/basic DESTINATION examples)
install(FILES docs/SDK.md DESTINATION . RENAME README.md)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-LICENSE.MIT)
if(WIN32)
  include(cmake/InstallRuntime.cmake)
endif()
